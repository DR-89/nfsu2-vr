# Assertions are executable checks in these tests, including fixture setup.
# Release builds must not silently remove them or their setup expressions.
get_property(community_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(community_target IN LISTS community_targets)
    if(community_target MATCHES "_test$")
        if(MSVC)
            target_compile_options(${community_target} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${community_target} PRIVATE -UNDEBUG)
        endif()
    endif()
endforeach()
