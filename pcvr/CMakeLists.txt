cmake_minimum_required(VERSION 3.24)
project(nfsu2_pcvr LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "PCVR requires a 64-bit Windows toolchain")
endif()
get_filename_component(NFSU2_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(NFSU2_DEPS_DIR "${NFSU2_ROOT}/build/deps" CACHE PATH "Source dependencies")
list(PREPEND CMAKE_MODULE_PATH "${NFSU2_ROOT}/android/cmake")
set(NFSU2_PCVR ON CACHE BOOL "Windows OpenXR host" FORCE)
set(NFSU2_VULKAN ON CACHE BOOL "" FORCE)
set(NFSU2_GEN_DIR "${NFSU2_ROOT}/gen" CACHE PATH "Generated game code")
add_compile_definitions(NFSU2_PCVR=1 NFSU2_OPENXR=1 _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES WIN32_LEAN_AND_MEAN NOMINMAX)
include_directories("${CMAKE_CURRENT_LIST_DIR}/compat" "${NFSU2_ROOT}/android/runtime")
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${NFSU2_DEPS_DIR}/SDL" "${CMAKE_BINARY_DIR}/SDL")
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${NFSU2_DEPS_DIR}/glslang" "${CMAKE_BINARY_DIR}/glslang")
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_LOADER ON CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER ON CACHE BOOL "" FORCE)
add_subdirectory("${NFSU2_DEPS_DIR}/OpenXR-SDK" "${CMAKE_BINARY_DIR}/openxr")
find_package(Vulkan REQUIRED)
add_library(nfsu2_xr STATIC "${NFSU2_ROOT}/android/runtime/quest_xr.c")
target_include_directories(nfsu2_xr PUBLIC "${NFSU2_ROOT}/android/runtime")
target_link_libraries(nfsu2_xr PUBLIC openxr_loader SDL2::SDL2 Vulkan::Vulkan)
add_subdirectory("${NFSU2_ROOT}" "${CMAKE_BINARY_DIR}/game")
# Share the Quest importer's metadata; the generated include contains no game bytes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NFSU2_ROOT}/android/app/src/main/java/org/nfsu2/quest/GameManifest.java")
file(STRINGS "${NFSU2_ROOT}/android/app/src/main/java/org/nfsu2/quest/GameManifest.java" game_manifest_lines REGEX "new GameDataImporter.Expected")
list(LENGTH game_manifest_lines game_manifest_count)
if(NOT game_manifest_count EQUAL 12)
    message(FATAL_ERROR "The supported game manifest must have exactly 12 entries")
endif()
set(game_manifest "")
foreach(line IN LISTS game_manifest_lines)
    string(REGEX REPLACE ".*Expected\\(\"([^\"]+)\", ([0-9]+)L, \"([0-9a-f]+)\"\\).*" "{\"\\1\", \\2ull, \"\\3\"},\n" entry "${line}")
    string(APPEND game_manifest "${entry}")
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/pcvr_game_manifest.inc" CONTENT "${game_manifest}")
add_library(pcvr_game_data STATIC game_data.cpp setup.cpp)
target_compile_features(pcvr_game_data PUBLIC cxx_std_17)
target_compile_options(pcvr_game_data PRIVATE /EHsc /utf-8)
target_compile_definitions(pcvr_game_data PRIVATE UNICODE _UNICODE)
target_include_directories(pcvr_game_data PUBLIC "${CMAKE_CURRENT_LIST_DIR}" PRIVATE "${CMAKE_BINARY_DIR}")
target_link_libraries(pcvr_game_data PUBLIC bcrypt comdlg32 comctl32 shell32 user32 gdi32)
target_link_libraries(nfsu2_recomp PRIVATE pcvr_game_data)
add_executable(pcvr_game_data_test "${NFSU2_ROOT}/tests/pcvr_game_data_test.cpp")
target_compile_options(pcvr_game_data_test PRIVATE /EHsc /utf-8)
target_link_libraries(pcvr_game_data_test PRIVATE pcvr_game_data)
target_link_libraries(xbox_nv2a_vk PUBLIC nfsu2_xr)
target_link_libraries(xbox_input PUBLIC nfsu2_xr)
target_link_libraries(xbox_kernel PRIVATE synchronization)
target_sources(nfsu2_recomp PRIVATE pcvr_host.c)
target_link_libraries(nfsu2_recomp PRIVATE nfsu2_xr)
add_executable(pcvr_multiview_test "${NFSU2_ROOT}/tests/quest_multiview_test.c")
target_link_libraries(pcvr_multiview_test PRIVATE Vulkan::Vulkan)
add_executable(pcvr_vertex_layout_test "${NFSU2_ROOT}/tests/quest_multiview_test.c")
target_compile_definitions(pcvr_vertex_layout_test PRIVATE NFSU2_VERTEX_LAYOUT_TEST=1)
target_link_libraries(pcvr_vertex_layout_test PRIVATE Vulkan::Vulkan)
add_executable(pcvr_culling_test "${NFSU2_ROOT}/tests/quest_culling_test.c")
add_executable(pcvr_shader_test "${NFSU2_ROOT}/tests/quest_shader_test.c"
    "${NFSU2_ROOT}/xboxrecomp/src/nv2a_gl/gl_vsh.c"
    "${NFSU2_ROOT}/xboxrecomp/src/nv2a_gl/gl_psh.c")
target_include_directories(pcvr_shader_test PRIVATE "${NFSU2_ROOT}/xboxrecomp/src/nv2a_gl")
add_executable(pcvr_fence_test "${NFSU2_ROOT}/tests/pcvr_fence_test.c")
add_executable(pcvr_surface_lookup_test "${NFSU2_ROOT}/tests/pcvr_surface_lookup_test.c")
add_executable(pcvr_bulk_shadow_test "${NFSU2_ROOT}/tests/pcvr_bulk_shadow_test.c")
add_executable(pcvr_indices_test "${NFSU2_ROOT}/tests/pcvr_indices_test.c")
add_executable(pcvr_index_packets_test "${NFSU2_ROOT}/tests/pcvr_index_packets_test.c")
add_executable(pcvr_texture_decode_test "${NFSU2_ROOT}/tests/pcvr_texture_decode_test.c")
add_executable(pcvr_dispatch_wait_test "${NFSU2_ROOT}/tests/pcvr_dispatch_wait_test.c")
target_include_directories(pcvr_dispatch_wait_test PRIVATE "${NFSU2_ROOT}/xboxrecomp/src")
add_executable(pcvr_pb_boundary_test "${NFSU2_ROOT}/tests/pcvr_pb_boundary_test.c"
    "${NFSU2_ROOT}/xboxrecomp/src/kernel/nv2a_pb_scan.c")
add_executable(pcvr_threads pcvr_threads.c)
target_link_libraries(pcvr_threads PRIVATE dbghelp)
add_executable(pcvr_cpu_sample pcvr_cpu_sample.c)
target_link_libraries(pcvr_cpu_sample PRIVATE dbghelp)
set_target_properties(nfsu2_recomp PROPERTIES WIN32_EXECUTABLE FALSE OUTPUT_NAME nfsu2-pcvr)
target_link_options(nfsu2_recomp PRIVATE /STACK:16777216 /DEBUG:FULL /OPT:REF /OPT:ICF /INCREMENTAL:NO)
file(GLOB pcvr_generated "${NFSU2_GEN_DIR}/*.c")
set_source_files_properties(${pcvr_generated} DIRECTORY "${NFSU2_ROOT}" PROPERTIES COMPILE_OPTIONS "/O1;/w")
set_source_files_properties("${NFSU2_ROOT}/src/recomp_manual.c" DIRECTORY "${NFSU2_ROOT}" PROPERTIES COMPILE_OPTIONS "/fp:strict")
add_custom_target(pcvr_bundle
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL2>" "$<TARGET_FILE_DIR:nfsu2_recomp>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:openxr_loader>" "$<TARGET_FILE_DIR:nfsu2_recomp>"
    DEPENDS nfsu2_recomp pcvr_game_data_test pcvr_multiview_test pcvr_vertex_layout_test pcvr_culling_test pcvr_shader_test pcvr_fence_test pcvr_surface_lookup_test pcvr_bulk_shadow_test pcvr_indices_test pcvr_index_packets_test pcvr_texture_decode_test pcvr_dispatch_wait_test pcvr_pb_boundary_test pcvr_threads)
